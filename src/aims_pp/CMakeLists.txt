cmake_minimum_required(VERSION 3.16)
project(aims_pp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(vesc_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenSSL REQUIRED)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(aims_pp_runtime SHARED src/runtime.cpp)
target_include_directories(aims_pp_runtime PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(aims_pp_runtime PRIVATE yaml-cpp OpenSSL::Crypto)
target_compile_options(aims_pp_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pp_node src/node.cpp)
target_link_libraries(pp_node aims_pp_runtime)
ament_target_dependencies(pp_node rclcpp rcl_interfaces ackermann_msgs diagnostic_msgs geometry_msgs nav_msgs std_msgs std_srvs vesc_msgs)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS aims_pp_runtime EXPORT aims_ppTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS pp_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md README.zh-CN.md LICENSE DESTINATION share/${PROJECT_NAME})
ament_export_targets(aims_ppTargets HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_libraries(aims_pp_runtime)
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 add_executable(pp_native_runner test/native_runner.cpp)
 target_link_libraries(pp_native_runner aims_pp_runtime)
 ament_add_pytest_test(pp_full_regression test TIMEOUT 120
   ENV AIMS_PP_NATIVE_RUNNER=$<TARGET_FILE:pp_native_runner> AIMS_PP_NATIVE_NODE=$<TARGET_FILE:pp_node> ROS_DOMAIN_ID=176
   APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_package()
