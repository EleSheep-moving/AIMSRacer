cmake_minimum_required(VERSION 3.16)
project(aims_mpcc_rt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenSSL REQUIRED)

add_library(mpcc_rt_core src/core.cpp src/execution.cpp)
target_include_directories(mpcc_rt_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_options(mpcc_rt_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(mpcc_rt_core PUBLIC yaml-cpp OpenSSL::Crypto ${CMAKE_DL_LIBS})
add_executable(replay_requests tools/replay_requests.cpp)
target_link_libraries(replay_requests mpcc_rt_core)
install(TARGETS replay_requests DESTINATION lib/${PROJECT_NAME})
add_executable(replay_runtime tools/replay_runtime.cpp)
target_link_libraries(replay_runtime mpcc_rt_core)
install(TARGETS replay_runtime DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  foreach(name health output history)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${name}.cpp")
      add_executable(test_${name} tests/test_${name}.cpp)
      # These pure contract tests use assert(), including state-changing calls.
      # Keep their checks active in the production Release test build.
      target_compile_options(test_${name} PRIVATE -UNDEBUG)
      target_include_directories(test_${name} PRIVATE include)
      add_test(NAME test_${name} COMMAND test_${name})
    endif()
  endforeach()
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE mpcc_rt_core)
  add_executable(test_bundle tests/test_bundle.cpp)
  target_link_libraries(test_bundle PRIVATE mpcc_rt_core)

  target_link_libraries(test_reanchor PRIVATE mpcc_rt_core)
  add_executable(test_execution tests/test_execution.cpp)
  target_link_libraries(test_execution PRIVATE mpcc_rt_core)
  add_executable(test_reference tests/test_reference.cpp)
  target_link_libraries(test_reference PRIVATE mpcc_rt_core)
  set(AIMS_MPCC_RT_TEST_BUNDLE "" CACHE PATH "Prepared native bundle for model and loop tests")
  set(AIMS_MPCC_RT_V2_TEST_BUNDLE "" CACHE PATH "Prepared rate_bounded_v2 bundle for execution tests")
  set(AIMS_MPCC_RT_MAP_TEST_BUNDLE "" CACHE PATH "Prepared map v2 bundle for atomic localization service tests")
  set(AIMS_MPCC_RT_UNQUALIFIED_TEST_BUNDLE "" CACHE PATH "Prepared v2 50 ms bundle for explicit startup rejection")
  if(AIMS_MPCC_RT_UNQUALIFIED_TEST_BUNDLE AND TARGET test_runtime_lifecycle)
    add_test(NAME test_runtime_mesh_rejection COMMAND test_runtime_lifecycle "${AIMS_MPCC_RT_UNQUALIFIED_TEST_BUNDLE}" mesh)
    set_tests_properties(test_runtime_mesh_rejection PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=230;ROS_LOCALHOST_ONLY=1")

  if(AIMS_MPCC_RT_V2_TEST_BUNDLE)
    add_test(NAME test_reference_v2 COMMAND test_reference "${AIMS_MPCC_RT_V2_TEST_BUNDLE}")
    add_test(NAME test_stage_zero_v2 COMMAND test_core "${AIMS_MPCC_RT_V2_TEST_BUNDLE}"
      --stage-zero-fixture "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/lead100_stage_zero.json")
    add_test(NAME test_reanchor_envelope_v2 COMMAND test_core "${AIMS_MPCC_RT_V2_TEST_BUNDLE}"
      --reanchor-envelope-fixture "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/reanchor_envelope.json")
    if(TARGET test_runtime_lifecycle)
      add_test(NAME test_runtime_cap_v2 COMMAND test_runtime_lifecycle "${AIMS_MPCC_RT_V2_TEST_BUNDLE}" cap)
      set_tests_properties(test_runtime_cap_v2 PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=230;ROS_LOCALHOST_ONLY=1")
    endif()
    add_test(NAME test_execution_v2 COMMAND test_execution "${AIMS_MPCC_RT_V2_TEST_BUNDLE}")
    add_test(NAME test_core_v2 COMMAND test_core "${AIMS_MPCC_RT_V2_TEST_BUNDLE}")
    add_test(NAME test_bundle_v2 COMMAND test_bundle "${AIMS_MPCC_RT_V2_TEST_BUNDLE}")
    add_test(NAME test_reanchor_v2 COMMAND test_reanchor "${AIMS_MPCC_RT_V2_TEST_BUNDLE}")
  endif()
  if(AIMS_MPCC_RT_MAP_TEST_BUNDLE AND TARGET test_runtime_lifecycle)
    add_test(NAME test_reference_map COMMAND test_reference "${AIMS_MPCC_RT_MAP_TEST_BUNDLE}")
    add_test(NAME test_runtime_map_anchor_order COMMAND test_runtime_lifecycle "${AIMS_MPCC_RT_MAP_TEST_BUNDLE}" anchor_order)

  if(AIMS_MPCC_RT_TEST_BUNDLE)
    add_test(NAME test_core COMMAND test_core "${AIMS_MPCC_RT_TEST_BUNDLE}")
    add_test(NAME test_bundle COMMAND test_bundle "${AIMS_MPCC_RT_TEST_BUNDLE}")
