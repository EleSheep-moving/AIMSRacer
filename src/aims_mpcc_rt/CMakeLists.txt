cmake_minimum_required(VERSION 3.16)
project(aims_mpcc_rt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenSSL REQUIRED)

add_library(mpcc_rt_core src/core.cpp)
target_include_directories(mpcc_rt_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_options(mpcc_rt_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(mpcc_rt_core PUBLIC yaml-cpp OpenSSL::Crypto ${CMAKE_DL_LIBS})
add_executable(replay_requests tools/replay_requests.cpp)
target_link_libraries(replay_requests mpcc_rt_core)
install(TARGETS replay_requests DESTINATION lib/${PROJECT_NAME})

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/node.cpp")
  find_package(rclcpp REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(ackermann_msgs REQUIRED)
  find_package(diagnostic_msgs REQUIRED)
  find_package(tf2 REQUIRED)
  find_package(tf2_ros REQUIRED)
  add_executable(mpcc_rt_node src/node.cpp)
  target_link_libraries(mpcc_rt_node mpcc_rt_core)
  ament_target_dependencies(mpcc_rt_node rclcpp nav_msgs geometry_msgs std_msgs
    std_srvs ackermann_msgs diagnostic_msgs tf2 tf2_ros)
  install(TARGETS mpcc_rt_node DESTINATION lib/${PROJECT_NAME})
endif()

include(CTest)
if(BUILD_TESTING)
  foreach(name health output history)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${name}.cpp")
      add_executable(test_${name} tests/test_${name}.cpp)
      # These pure contract tests use assert(), including state-changing calls.
      # Keep their checks active in the production Release test build.
      target_compile_options(test_${name} PRIVATE -UNDEBUG)
      target_include_directories(test_${name} PRIVATE include)
      add_test(NAME test_${name} COMMAND test_${name})
    endif()
  endforeach()
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE mpcc_rt_core)
  add_executable(test_bundle tests/test_bundle.cpp)
  target_link_libraries(test_bundle PRIVATE mpcc_rt_core)
  add_executable(test_reanchor tests/test_reanchor.cpp)
  target_link_libraries(test_reanchor PRIVATE mpcc_rt_core)
  set(AIMS_MPCC_RT_TEST_BUNDLE "" CACHE PATH "Prepared native bundle for model and loop tests")
  if(AIMS_MPCC_RT_TEST_BUNDLE)
    add_test(NAME test_core COMMAND test_core "${AIMS_MPCC_RT_TEST_BUNDLE}")
    add_test(NAME test_bundle COMMAND test_bundle "${AIMS_MPCC_RT_TEST_BUNDLE}")
    add_test(NAME test_reanchor COMMAND test_reanchor "${AIMS_MPCC_RT_TEST_BUNDLE}")
  endif()
endif()

install(TARGETS mpcc_rt_core EXPORT mpcc_rt_coreTargets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(FILES NOTICE.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY licenses DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/export_bundle.py scripts/build_bundle.py DESTINATION lib/${PROJECT_NAME})
ament_export_targets(mpcc_rt_coreTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(yaml-cpp OpenSSL)
ament_package()
