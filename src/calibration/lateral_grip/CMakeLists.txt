cmake_minimum_required(VERSION 3.16)
project(aims_lateral_grip)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(vesc_msgs REQUIRED)
find_package(crsf_receiver_msg REQUIRED)
find_package(OpenSSL REQUIRED)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(lateral_grip_calib src/node.cpp)
target_include_directories(lateral_grip_calib PRIVATE include ${NLOHMANN_JSON_INCLUDE_DIR})
ament_target_dependencies(lateral_grip_calib rclcpp rcl_interfaces std_msgs std_srvs nav_msgs geometry_msgs sensor_msgs diagnostic_msgs vesc_msgs crsf_receiver_msg)
target_link_libraries(lateral_grip_calib OpenSSL::Crypto)
install(TARGETS lateral_grip_calib DESTINATION lib/${PROJECT_NAME})
foreach(operation inspect start stop)
  add_executable(${operation} src/operator.cpp)
  target_compile_definitions(${operation} PRIVATE LATERAL_OPERATION="${operation}")
  ament_target_dependencies(${operation} rclcpp std_srvs diagnostic_msgs)
  install(TARGETS ${operation} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(PROGRAMS scripts/analyze.py DESTINATION lib/${PROJECT_NAME} RENAME analyze)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md README.zh-CN.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  add_executable(test_lateral_core tests/test_core.cpp)
  target_include_directories(test_lateral_core PRIVATE include)
  target_compile_options(test_lateral_core PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  add_test(NAME lateral_core COMMAND test_lateral_core)
  add_executable(test_lateral_evidence tests/test_evidence.cpp)
  target_include_directories(test_lateral_evidence PRIVATE include)
  target_compile_options(test_lateral_evidence PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  add_test(NAME lateral_evidence COMMAND test_lateral_evidence)
  ament_add_pytest_test(lateral_analysis tests/test_analysis.py)
  ament_add_pytest_test(lateral_node_process tests/test_node_process.py TIMEOUT 60 ENV ROS_DOMAIN_ID=227 ROS_LOCALHOST_ONLY=1 LATERAL_NODE_EXECUTABLE=$<TARGET_FILE:lateral_grip_calib> LATERAL_OPERATOR_DIR=$<TARGET_FILE_DIR:start> LATERAL_TEST_ARTIFACT_ROOT=${CMAKE_CURRENT_BINARY_DIR}/process_evidence)
endif()
ament_package()
