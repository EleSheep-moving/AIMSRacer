cmake_minimum_required(VERSION 3.16)
project(aims_calibration)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(vesc_msgs REQUIRED)
find_package(crsf_receiver_msg REQUIRED)
find_package(aims_pp REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_library(calibration_runtime src/native_runtime.cpp src/native_engine.cpp)
target_include_directories(calibration_runtime PUBLIC include ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(calibration_runtime PUBLIC Threads::Threads)
add_executable(longitudinal_calibration src/native_node.cpp)
ament_target_dependencies(longitudinal_calibration rclcpp std_msgs std_srvs diagnostic_msgs nav_msgs geometry_msgs rcl_interfaces vesc_msgs crsf_receiver_msg aims_pp)
target_include_directories(longitudinal_calibration PRIVATE include)
target_link_libraries(longitudinal_calibration calibration_runtime OpenSSL::Crypto)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS longitudinal_calibration DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md README.zh-CN.md DESTINATION share/${PROJECT_NAME})
foreach(command baseline plan results)
  configure_file(scripts/cli.in ${CMAKE_CURRENT_BINARY_DIR}/${command} @ONLY)
  install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/${command} DESTINATION lib/${PROJECT_NAME})
endforeach()
foreach(action fit query)
  configure_file(scripts/wheel_cli.in ${CMAKE_CURRENT_BINARY_DIR}/wheel-current-${action} @ONLY)
  install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/wheel-current-${action} DESTINATION lib/${PROJECT_NAME})
endforeach()
foreach(command inspect start stop)
  add_executable(${command} src/native_operator.cpp)
  target_include_directories(${command} PRIVATE include)
  ament_target_dependencies(${command} rclcpp diagnostic_msgs std_srvs nav_msgs)
  target_link_libraries(${command} calibration_runtime)
  install(TARGETS ${command} DESTINATION lib/${PROJECT_NAME})
endforeach()
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_executable(test_native_runtime tests/test_native_runtime.cpp src/native_runtime.cpp)
  target_include_directories(test_native_runtime PRIVATE include ${NLOHMANN_JSON_INCLUDE_DIR})
  target_link_libraries(test_native_runtime Threads::Threads)
  target_compile_options(test_native_runtime PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  add_test(NAME native_runtime_contracts COMMAND test_native_runtime)
  add_executable(native_engine_trace tests/native_engine_trace.cpp)
  target_link_libraries(native_engine_trace calibration_runtime)
  add_test(NAME native_engine_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native_engine_parity.py)
  set_tests_properties(native_engine_parity PROPERTIES
    ENVIRONMENT "NATIVE_ENGINE_TRACE=$<TARGET_FILE:native_engine_trace>;PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:$ENV{PYTHONPATH}" TIMEOUT 180)
  ament_add_pytest_test(native_node_process tests/test_native_node_process.py TIMEOUT 120
    ENV ROS_DOMAIN_ID=218 ROS_LOCALHOST_ONLY=1 NATIVE_CALIBRATION_EXECUTABLE=$<TARGET_FILE:longitudinal_calibration>
    APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  ament_add_pytest_test(native_operator_runtime tests/test_native_operator_runtime.py TIMEOUT 60
    ENV ROS_DOMAIN_ID=219 ROS_LOCALHOST_ONLY=1 CALIB_NATIVE_OPERATOR_DIR=$<TARGET_FILE_DIR:start>
    APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  foreach(reference_test engine full_step_engine configuration full_step_configuration planner full_step_planner io full_step_io operator steady predictive_exit package cli wheel_current_model)
    ament_add_pytest_test(reference_${reference_test} tests/test_${reference_test}.py TIMEOUT 120
      APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
  endforeach()
endif()
ament_package()
