cmake_minimum_required(VERSION 3.16)
project(aims_mocap_validation)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(vesc_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(mocap_capture src/capture_node.cpp)
target_compile_features(mocap_capture PRIVATE cxx_std_17)
target_compile_options(mocap_capture PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(mocap_capture rclcpp sensor_msgs nav_msgs geometry_msgs diagnostic_msgs vesc_msgs ackermann_msgs)
target_link_libraries(mocap_capture OpenSSL::Crypto)
install(TARGETS mocap_capture DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/mocap DESTINATION lib/${PROJECT_NAME})
install(FILES ../sources.lock.json DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES aims_mocap_validation/config.template.json DESTINATION share/${PROJECT_NAME}/config)
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 find_package(VRPN REQUIRED)
 add_executable(vrpn_fixture_server tests/vrpn_fixture_server.cpp)
 target_include_directories(vrpn_fixture_server PRIVATE ${VRPN_INCLUDE_DIRS})
 target_link_libraries(vrpn_fixture_server ${VRPN_LIBRARIES})
 ament_add_pytest_test(mocap_tools tests TIMEOUT 180 ENV "MOCAP_CAPTURE_EXECUTABLE=$<TARGET_FILE:mocap_capture>" "MOCAP_VRPN_FIXTURE=$<TARGET_FILE:vrpn_fixture_server>" "ROS_DOMAIN_ID=212" "ROS_LOCALHOST_ONLY=1" "OPENBLAS_NUM_THREADS=1")
endif()
ament_package()
